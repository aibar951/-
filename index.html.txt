function scoreColor(score) {
  if (score >= 85) return "#10B981";
  if (score >= 70) return "#F59E0B";
  return "#EF4444";
}
 
function scoreLabel(score) {
  if (score >= 85) return "Отлично";
  if (score >= 70) return "Хорошо";
  return "Плохо";
}
 
function CircularScore({ score }) {
  const r = 80;
  const circ = 2 * Math.PI * r;
  const pct = score / 100;
  const color = scoreColor(score);
  return (
    <div className="relative flex items-center justify-center" style={{ width: 220, height: 220 }}>
      <svg width="220" height="220" style={{ position: "absolute", top: 0, left: 0 }}>
        <circle cx="110" cy="110" r={r} fill="none" stroke="#1E293B" strokeWidth="14" />
        <circle
          cx="110" cy="110" r={r}
          fill="none"
          stroke={color}
          strokeWidth="14"
          strokeDasharray={circ}
          strokeDashoffset={circ * (1 - pct)}
          strokeLinecap="round"
          transform="rotate(-90 110 110)"
          style={{ filter: `drop-shadow(0 0 10px ${color}88)`, transition: "stroke-dashoffset 1.2s cubic-bezier(.4,0,.2,1)" }}
        />
      </svg>
      <div className="flex flex-col items-center">
        <span className="text-6xl font-bold" style={{ color, textShadow: `0 0 20px ${color}66`, fontFamily: "'DM Mono', monospace" }}>{score}</span>
        <span className="text-xs font-semibold tracking-widest mt-1" style={{ color, opacity: 0.85 }}>{scoreLabel(score).toUpperCase()}</span>
        <span className="text-xs text-slate-500 mt-1">из 100</span>
      </div>
    </div>
  );
}

function BarChartMini({ data }) {
  const max = Math.max(...data.map(d => d.score));
  return (
    <div className="flex items-end gap-2 h-40">
      {data.map((d, i) => {
        const isLast = i === data.length - 1;
        const color = scoreColor(d.score);
        const h = (d.score / max) * 100;
        return (
          <div key={d.day} className="flex-1 flex flex-col items-center gap-1">
            <div className="text-xs font-mono" style={{ color, fontSize: 11 }}>{d.score}</div>
            <div className="w-full relative flex-1 flex items-end rounded-t">

                }}
              />
            </div>
            <div className="text-xs text-slate-500">{d.day}</div>
          </div>
        );
      })}
    </div>
  );
}

function AlarmView({ blueLightFilter, setBlueLightFilter }) {
  const [wakeHour, setWakeHour] = useState(7);
  const [wakeMin, setWakeMin] = useState(0);
 
  const sleepCycles = [1, 2, 3, 4, 5].map(cycles => {
    const totalMin = cycles * 90 + 15;
    const wakeTotal = wakeHour * 60 + wakeMin;
    let bedTotal = wakeTotal - totalMin;
    if (bedTotal < 0) bedTotal += 24 * 60;
    const h = Math.floor(bedTotal / 60) % 24;
    const m = bedTotal % 60;
    return { cycles, h, m: m.toString().padStart(2, "0") };
  }).reverse();
 
  const fmtWake = `${wakeHour.toString().padStart(2, "0")}:${wakeMin.toString().padStart(2, "0")}`;
 
  return (
    <div className="flex flex-col gap-6">
      {/* Wake time setter */}
      <div className="rounded-2xl p-5 border" style={{ background: "#161E35", borderColor: "#1E293B" }}>
        <div className="text-white font-bold mb-1 flex items-center gap-2">
          <Bell size={16} className="text-cyan-400" />
          Время подъёма
        </div>
        <div className="text-slate-500 text-xs mb-4">Установите желаемое время будильника</div>
        <div className="flex items-center gap-3 mb-4">
          <div className="flex-1">
